# 五子棋 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a 15×15 free-style Gomoku game with on-device AI opponent, deployed on Cloudflare Pages, plus a landing page listing games.

**Architecture:** Static site (HTML/CSS/JS, no build step) deployed via Wrangler to Cloudflare Pages. AI runs in a Web Worker so the UI never freezes. Each game lives in its own folder; the landing page renders a card grid from a data array.

**Tech Stack:** Plain HTML/CSS/JavaScript (ES modules), HTML5 Canvas, Web Workers, Node's built-in test runner, Wrangler for Cloudflare Pages.

## Global Constraints

- **No runtime npm dependencies.** Only `wrangler` as devDependency.
- **No build step.** Source files are served as-is.
- **15×15 free-style Gomoku rules** (no forbidden moves).
- **Player is Black and plays first.**
- **Three difficulties**: easy (depth 2), medium (depth 4), hard (depth 6).
- **All file paths in this plan are relative to `D:\play\happygames`**.
- **Commands assume PowerShell on Windows** unless prefixed with `bash`.
- **AI evaluation function pattern scores** (from spec): 连5 ±100000, 活4 ±10000, 冲4 ±1000, 活3 ±1000, 眠3 ±100, 活2 ±100, 眠2 ±10, 活1 ±1.
- **Difficulty config**: easy `{depth: 2, candidateRadius: 1, defenseWeight: 0.6}`, medium `{depth: 4, candidateRadius: 2, defenseWeight: 1.0}`, hard `{depth: 6, candidateRadius: 2, defenseWeight: 1.2}`.
- **Web Worker import**: `ai.worker.js` must use ES module syntax and be loaded with `{ type: 'module' }` from the page.

Write to `D:\play\happygames\gomoku\ai.js`:

```js
// Gomoku AI — pure functions, no DOM.
// Conventions: BLACK = 1 (player), WHITE = 2 (AI).
// Board is a 2D array [y][x] where EMPTY = 0.

export const SIZE = 15;
export const EMPTY = 0;
export const BLACK = 1;
export const WHITE = 2;

export const OPPOSITE = { [BLACK]: WHITE, [WHITE]: BLACK };

export function createBoard() {
  return Array.from({ length: SIZE }, () => Array(SIZE).fill(EMPTY));
}

export function inBounds(x, y) {
  return x >= 0 && x < SIZE && y >= 0 && y < SIZE;
}

export function place(board, x, y, color) {
  board[y][x] = color;
}

export function unplace(board, x, y) {
  board[y][x] = EMPTY;
}

// Pattern scores (from spec)
const SCORE = {
  FIVE: 100000,
  OPEN_FOUR: 10000,
  CLOSED_FOUR: 1000,
  OPEN_THREE: 1000,
  CLOSED_THREE: 100,
  OPEN_TWO: 100,
  CLOSED_TWO: 10,
  ONE: 1,
};

const DIRS = [
  [1, 0],   // horizontal
  [0, 1],   // vertical
  [1, 1],   // diagonal down-right
  [1, -1],  // diagonal down-left
];

// Walk a line through (x,y) in direction (dx,dy), counting the run of `color`
// stones and noting whether each end is blocked.
function analyzeLine(board, x, y, dx, dy, color) {
  // Count consecutive `color` stones starting at (x,y), including (x,y).
  let count = 1;
  let leftBlocked = false;
  let rightBlocked = false;

  let nx = x + dx, ny = y + dy;
  while (inBounds(nx, ny) && board[ny][nx] === color) {
    count++; nx += dx; ny += dy;
  }
  if (!inBounds(nx, ny) || board[ny][nx] === OPPOSITE[color]) {
    rightBlocked = true;
  }

  nx = x - dx; ny = y - dy;
  while (inBounds(nx, ny) && board[ny][nx] === color) {
    count++; nx -= dx; ny -= dy;
  }
  if (!inBounds(nx, ny) || board[ny][nx] === OPPOSITE[color]) {
    leftBlocked = true;
  }

  return { count, leftBlocked, rightBlocked };
}

// Score a single line's contribution. The cell (x,y) itself is `color`.
function lineScore(count, leftBlocked, rightBlocked) {
  if (count >= 5) return SCORE.FIVE;
  const bothOpen = !leftBlocked && !rightBlocked;
  const oneOpen = !leftBlocked || !rightBlocked;

  switch (count) {
    case 4:
      if (bothOpen) return SCORE.OPEN_FOUR;
      if (oneOpen) return SCORE.CLOSED_FOUR;
      return 0;
    case 3:
      if (bothOpen) return SCORE.OPEN_THREE;
      if (oneOpen) return SCORE.CLOSED_THREE;
      return 0;
    case 2:
      if (bothOpen) return SCORE.OPEN_TWO;
      if (oneOpen) return SCORE.CLOSED_TWO;
      return 0;
    case 1:
      if (bothOpen) return SCORE.ONE;
      return 0;
    default:
      return 0;
  }
}

// Compute total pattern score for `color` on the whole board.
// To avoid double-counting a run of N stones, only count the cell that is
// the "leftmost" or "bottom" endpoint of the run for each direction.
// We do this by checking that the previous cell in the line is empty or
// opposite (i.e., a boundary for the run).
function scoreForColor(board, color) {
  let total = 0;
  for (let y = 0; y < SIZE; y++) {
    for (let x = 0; x < SIZE; x++) {
      if (board[y][x] !== color) continue;
      for (const [dx, dy] of DIRS) {
        const px = x - dx, py = y - dy;
        const prev = inBounds(px, py) ? board[py][px] : OPPOSITE[color];
        // Only count this stone if it's the start of a run in this direction.
        if (prev === color) continue;
        const { count, leftBlocked, rightBlocked } = analyzeLine(board, x, y, dx, dy, color);
        total += lineScore(count, leftBlocked, rightBlocked);
      }
    }
  }
  return total;
}

// Total evaluation for `color`: my score minus opponent's score times defense weight.
export function evaluate(board, color, defenseWeight = 1.0) {
  const me = scoreForColor(board, color);
  const opp = scoreForColor(board, OPPOSITE[color]);
  return me - opp * defenseWeight;
}

export function isFive(board, x, y, color) {
  if (board[y][x] !== color) return false;
  for (const [dx, dy] of DIRS) {
    const { count } = analyzeLine(board, x, y, dx, dy, color);
    if (count >= 5) return true;
  }
  return false;
}
```

export const DIFFICULTY = {
  easy:   { depth: 2, candidateRadius: 1, defenseWeight: 0.6 },
  medium: { depth: 4, candidateRadius: 2, defenseWeight: 1.0 },
  hard:   { depth: 6, candidateRadius: 2, defenseWeight: 1.2 },
};

// Generate candidate moves: empty cells within `radius` of any existing stone.
// Order: cells closest to the densest cluster first (we'll re-order by score next).
export function generateCandidates(board, radius) {
  const seen = new Set();
  const out = [];
  for (let y = 0; y < SIZE; y++) {
    for (let x = 0; x < SIZE; x++) {
      if (board[y][x] === EMPTY) continue;
      for (let dy = -radius; dy <= radius; dy++) {
        for (let dx = -radius; dx <= radius; dx++) {
          const nx = x + dx, ny = y + dy;
          if (!inBounds(nx, ny)) continue;
          if (board[ny][nx] !== EMPTY) continue;
          const key = nx * SIZE + ny;
          if (seen.has(key)) continue;
          seen.add(key);
          out.push({ x: nx, y: ny });
        }
      }
    }
  }
  return out;
}

// Estimate the value of placing `color` at (x,y), without committing.
function singleMoveValue(board, x, y, color, defenseWeight) {
  board[y][x] = color;
  const v = evaluate(board, color, defenseWeight);
  board[y][x] = EMPTY;
  return v;
}

function orderMoves(board, candidates, color, defenseWeight) {
  return candidates
    .map(m => ({ ...m, _h: singleMoveValue(board, m.x, m.y, color, defenseWeight) }))
    .sort((a, b) => b._h - a._h);
}

function minimax(board, depth, alpha, beta, maximizing, color, defenseWeight) {
  // Terminal: no candidates means board full → draw, score 0.
  const cands = generateCandidates(board, 1);
  if (depth === 0 || cands.length === 0) {
    return evaluate(board, color, defenseWeight);
  }
  const ordered = orderMoves(board, cands, maximizing ? color : OPPOSITE[color], defenseWeight);

  if (maximizing) {
    let value = -Infinity;
    for (const m of ordered) {
      place(board, m.x, m.y, color);
      const v = minimax(board, depth - 1, alpha, beta, false, color, defenseWeight);
      unplace(board, m.x, m.y);
      if (v > value) value = v;
      if (value > alpha) alpha = value;
      if (alpha >= beta) break;
    }
    return value;
  } else {
    let value = Infinity;
    const opp = OPPOSITE[color];
    for (const m of ordered) {
      place(board, m.x, m.y, opp);
      const v = minimax(board, depth - 1, alpha, beta, true, color, defenseWeight);
      unplace(board, m.x, m.y);
      if (v < value) value = v;
      if (value < beta) beta = value;
      if (alpha >= beta) break;
    }
    return value;
  }
}

export function bestMove(board, color, difficulty) {
  const { depth, candidateRadius, defenseWeight } = difficulty;

  // Empty board shortcut.
  let hasAny = false;
  for (let y = 0; y < SIZE && !hasAny; y++) {
    for (let x = 0; x < SIZE && !hasAny; x++) {
      if (board[y][x] !== EMPTY) hasAny = true;
    }
  }
  if (!hasAny) return { x: 7, y: 7, score: 0 };

  const candidates = generateCandidates(board, candidateRadius);
  if (candidates.length === 0) return null; // board full

  // Immediate winning move check.
  for (const m of candidates) {
    place(board, m.x, m.y, color);
    const win = isFive(board, m.x, m.y, color);
    unplace(board, m.x, m.y);
    if (win) return { x: m.x, y: m.y, score: Infinity };
  }

  // Block opponent's immediate winning move.
  const opp = OPPOSITE[color];
  for (const m of candidates) {
    place(board, m.x, m.y, opp);
    const oppWin = isFive(board, m.x, m.y, opp);
    unplace(board, m.x, m.y);
    if (oppWin) return { x: m.x, y: m.y, score: Infinity };
  }

  const ordered = orderMoves(board, candidates, color, defenseWeight);
  let best = null;
  let alpha = -Infinity;
  const beta = Infinity;

  for (const m of ordered) {
    place(board, m.x, m.y, color);
    const v = minimax(board, depth - 1, alpha, beta, false, color, defenseWeight);
    unplace(board, m.x, m.y);
    if (best === null || v > best.score) {
      best = { x: m.x, y: m.y, score: v };
    }
    if (v > alpha) alpha = v;
  }

  return best;
}

import { SIZE, BLACK, WHITE, EMPTY } from '/gomoku/ai.js';

const PADDING_RATIO = 0.04;  // space around the grid
const STAR_POINTS = [
  [3, 3], [3, 11], [11, 3], [11, 11], [7, 7],
];

export function computeMetrics(size) {
  const padding = size * PADDING_RATIO;
  const gridSize = size - padding * 2;
  const cell = gridSize / (SIZE - 1);
  return { padding, cell, gridSize };
}

export function pixelToCell(metrics, px, py, size) {
  const x = Math.round((px - metrics.padding) / metrics.cell);
  const y = Math.round((py - metrics.padding) / metrics.cell);
  if (x < 0 || x >= SIZE || y < 0 || y >= SIZE) return null;
  return { x, y };
}

export function render(ctx, state) {
  const { board, lastMove, hoverMove, size, animatedStones } = state;
  const dpr = window.devicePixelRatio || 1;
  if (ctx.canvas.width !== size * dpr) {
    ctx.canvas.width = size * dpr;
    ctx.canvas.height = size * dpr;
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  }
  ctx.clearRect(0, 0, size, size);

  const m = computeMetrics(size);

  // Grid lines
  ctx.strokeStyle = '#6e5a40';
  ctx.lineWidth = 1;
  for (let i = 0; i < SIZE; i++) {
    const p = m.padding + i * m.cell;
    ctx.beginPath();
    ctx.moveTo(m.padding, p);
    ctx.lineTo(m.padding + m.gridSize, p);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(p, m.padding);
    ctx.lineTo(p, m.padding + m.gridSize);
    ctx.stroke();
  }

  // Star points
  ctx.fillStyle = '#6e5a40';
  for (const [sx, sy] of STAR_POINTS) {
    ctx.beginPath();
    ctx.arc(m.padding + sx * m.cell, m.padding + sy * m.cell, 3, 0, Math.PI * 2);
    ctx.fill();
  }

  // Stones
  const radius = m.cell * 0.44;
  for (let y = 0; y < SIZE; y++) {
    for (let x = 0; x < SIZE; x++) {
      const color = board[y][x];
      if (color === EMPTY) continue;
      const cx = m.padding + x * m.cell;
      const cy = m.padding + y * m.cell;
      const animated = animatedStones && animatedStones.get(`${x},${y}`);
      const scale = animated ? animated.scale : 1;
      drawStone(ctx, cx, cy, radius * scale, color);
    }
  }

  // Hover preview
  if (hoverMove && board[hoverMove.y][hoverMove.x] === EMPTY) {
    const cx = m.padding + hoverMove.x * m.cell;
    const cy = m.padding + hoverMove.y * m.cell;
    ctx.globalAlpha = 0.35;
    drawStone(ctx, cx, cy, radius, BLACK);
    ctx.globalAlpha = 1;
  }

  // Last-move marker
  if (lastMove && board[lastMove.y][lastMove.x] !== EMPTY) {
    const cx = m.padding + lastMove.x * m.cell;
    const cy = m.padding + lastMove.y * m.cell;
    ctx.fillStyle = '#c0392b';
    ctx.beginPath();
    ctx.arc(cx, cy, Math.max(3, radius * 0.2), 0, Math.PI * 2);
    ctx.fill();
  }
}

function drawStone(ctx, cx, cy, r, color) {
  const grad = ctx.createRadialGradient(cx - r * 0.3, cy - r * 0.3, r * 0.1, cx, cy, r);
  if (color === BLACK) {
    grad.addColorStop(0, '#555');
    grad.addColorStop(1, '#000');
  } else {
    grad.addColorStop(0, '#fff');
    grad.addColorStop(1, '#cfc7b3');
  }
  ctx.fillStyle = grad;
  ctx.beginPath();
  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  ctx.fill();
  ctx.strokeStyle = color === BLACK ? '#000' : '#999';
  ctx.lineWidth = 0.5;
  ctx.stroke();
}

import {
  createBoard, place, isFive, BLACK, WHITE, EMPTY, SIZE,
  OPPOSITE, DIFFICULTY,
} from '/gomoku/ai.js';
import { render, pixelToCell, computeMetrics } from '/gomoku/render.js';

const canvas = document.getElementById('board');
const statusEl = document.getElementById('status');
const resetBtn = document.getElementById('reset');
const diffButtons = document.querySelectorAll('.diff-btn');

const state = {
  board: createBoard(),
  lastMove: null,
  hoverMove: null,
  currentPlayer: BLACK,  // player (Black) goes first
  difficulty: 'medium',
  gameOver: false,
  aiThinking: false,
  worker: null,
  pendingRequestId: 0,
  animatedStones: new Map(),
  resizeObserver: null,
};

function init() {
  setupCanvas();
  setupWorker();
  setupEvents();
  draw();
  setStatus('轮到你了');
}

function setupCanvas() {
  const size = () => Math.min(canvas.parentElement.clientWidth, 600);
  state.resizeFn = () => {
    state.size = size();
    draw();
  };
  state.resizeFn();
  state.resizeObserver = new ResizeObserver(state.resizeFn);
  state.resizeObserver.observe(canvas.parentElement);
}

function setupWorker() {
  state.worker = new Worker('/gomoku/ai.worker.js', { type: 'module' });
  state.worker.onmessage = (e) => {
    const { ok, move, error, requestId } = e.data;
    if (requestId !== state.pendingRequestId) return; // stale
    state.aiThinking = false;
    canvas.style.pointerEvents = 'auto';
    if (!ok) {
      setStatus('AI 出错了，请重开', true);
      return;
    }
    if (!move) {
      setStatus('平局', true);
      state.gameOver = true;
      draw();
      return;
    }
    applyMove(move.x, move.y, WHITE);
    if (isFive(state.board, move.x, move.y, WHITE)) {
      state.gameOver = true;
      setStatus('电脑赢了', true);
    } else if (isBoardFull()) {
      state.gameOver = true;
      setStatus('平局', true);
    } else {
      setStatus('轮到你了');
      state.currentPlayer = BLACK;
    }
    draw();
  };
  state.worker.onerror = () => {
    state.aiThinking = false;
    setStatus('AI 出错了，请重开', true);
  };
}

function setupEvents() {
  canvas.addEventListener('click', onCanvasClick);
  canvas.addEventListener('mousemove', onCanvasMove);
  canvas.addEventListener('mouseleave', () => {
    state.hoverMove = null;
    draw();
  });
  resetBtn.addEventListener('click', () => reset());
  diffButtons.forEach(btn => {
    btn.addEventListener('click', () => onDifficultyClick(btn.dataset.difficulty));
  });
}

function onCanvasClick(e) {
  if (state.gameOver || state.aiThinking) return;
  if (state.currentPlayer !== BLACK) return;
  const rect = canvas.getBoundingClientRect();
  const px = e.clientX - rect.left;
  const py = e.clientY - rect.top;
  const m = computeMetrics(state.size);
  const cell = pixelToCell(m, px, py, state.size);
  if (!cell) return;
  if (state.board[cell.y][cell.x] !== EMPTY) return;
  applyMove(cell.x, cell.y, BLACK);
  if (isFive(state.board, cell.x, cell.y, BLACK)) {
    state.gameOver = true;
    setStatus('你赢了', true);
    draw();
    return;
  }
  if (isBoardFull()) {
    state.gameOver = true;
    setStatus('平局', true);
    draw();
    return;
  }
  state.currentPlayer = WHITE;
  setStatus('电脑思考中...');
  draw();
  requestAiMove();
}

function onCanvasMove(e) {
  if (state.gameOver || state.aiThinking) return;
  if (state.currentPlayer !== BLACK) return;
  const rect = canvas.getBoundingClientRect();
  const px = e.clientX - rect.left;
  const py = e.clientY - rect.top;
  const m = computeMetrics(state.size);
  const cell = pixelToCell(m, px, py, state.size);
  if (!cell) { state.hoverMove = null; draw(); return; }
  if (state.board[cell.y][cell.x] !== EMPTY) { state.hoverMove = null; draw(); return; }
  state.hoverMove = cell;
  draw();
}

function applyMove(x, y, color) {
  place(state.board, x, y, color);
  state.lastMove = { x, y };
  state.hoverMove = null;
  animateStone(x, y);
}

function animateStone(x, y) {
  const key = `${x},${y}`;
  const start = performance.now();
  const duration = 180;
  const tick = (now) => {
    const t = Math.min(1, (now - start) / duration);
    const scale = 0.3 + 0.7 * (1 - Math.pow(1 - t, 3)); // ease-out cubic
    state.animatedStones.set(key, { scale });
    draw();
    if (t < 1) {
      requestAnimationFrame(tick);
    } else {
      state.animatedStones.delete(key);
      draw();
    }
  };
  requestAnimationFrame(tick);
}

function isBoardFull() {
  for (let y = 0; y < SIZE; y++) {
    for (let x = 0; x < SIZE; x++) {
      if (state.board[y][x] === EMPTY) return false;
    }
  }
  return true;
}

function requestAiMove() {
  state.aiThinking = true;
  canvas.style.pointerEvents = 'none';
  state.pendingRequestId += 1;
  // Send a deep clone so the worker has a stable snapshot.
  const snapshot = state.board.map(row => row.slice());
  state.worker.postMessage({
    board: snapshot,
    color: WHITE,
    difficulty: state.difficulty,
    requestId: state.pendingRequestId,
  });
}

function setStatus(text, isWin = false) {
  statusEl.textContent = text;
  statusEl.classList.toggle('win', isWin);
}

function draw() {
  render(canvas.getContext('2d'), {
    board: state.board,
    lastMove: state.lastMove,
    hoverMove: state.hoverMove,
    size: state.size,
    animatedStones: state.animatedStones,
  });
}

function reset() {
  state.board = createBoard();
  state.lastMove = null;
  state.hoverMove = null;
  state.currentPlayer = BLACK;
  state.gameOver = false;
  state.aiThinking = false;
  state.animatedStones.clear();
  canvas.style.pointerEvents = 'auto';
  setStatus('轮到你了');
  draw();
}

function onDifficultyClick(newDifficulty) {
  if (newDifficulty === state.difficulty) return;
  const inProgress = state.lastMove !== null && !state.gameOver;
  if (inProgress) {
    const ok = window.confirm('重置当前对局？');
    if (!ok) return;
  }
  state.difficulty = newDifficulty;
  diffButtons.forEach(btn => {
    const isActive = btn.dataset.difficulty === newDifficulty;
    btn.classList.toggle('active', isActive);
    btn.setAttribute('aria-checked', String(isActive));
  });
  reset();
}

init();

    );
  }

test('head-to-head: hard beats easy in 20 AI-vs-AI games', { skip: false }, () => {
  let hardWins = 0, easyWins = 0, draws = 0;

  // Deterministic but varied openings.
  const openings = [
    [{ x: 7, y: 7, color: BLACK }],
    [{ x: 7, y: 7, color: BLACK }, { x: 8, y: 8, color: WHITE }],
    [{ x: 7, y: 7, color: BLACK }, { x: 8, y: 7, color: WHITE }, { x: 7, y: 8, color: BLACK }],
  ];

  for (let i = 0; i < 20; i++) {
    const b = createBoard();
    const opening = openings[i % openings.length];
    for (const m of opening) place(b, m.x, m.y, m.color);

    let hardColor = i % 2 === 0 ? BLACK : WHITE;
    let easyColor = OPPOSITE_MAP[hardColor];
    let turn = (opening.length % 2 === 0) ? BLACK : WHITE;
    let winner = null;

    // Cap moves to avoid infinite loops in case of bug.
    for (let moveCount = 0; moveCount < 225; moveCount++) {
      const moverColor = turn;
      const moverDiff = moverColor === hardColor ? DIFFICULTY.hard : DIFFICULTY.easy;
      const move = bestMove(b, moverColor, moverDiff);
      if (!move) { winner = 'draw'; break; }
      place(b, move.x, move.y, moverColor);
      if (isFive(b, move.x, move.y, moverColor)) { winner = moverColor; break; }
      turn = OPPOSITE_MAP[turn];
    }
    if (winner === null) winner = 'draw';

    if (winner === hardColor) hardWins++;
    else if (winner === easyColor) easyWins++;
    else draws++;
  }

  // Spec says: "hard wins >= easy wins in the AI-vs-AI matchup".
  // We require strict >, since hard is supposed to be strictly stronger; allow
  // a tolerance for draws (when neither side can win, neutral).
  assert.ok(
    hardWins >= easyWins,
    `hard should win >= easy across 20 games; got hard=${hardWins}, easy=${easyWins}, draws=${draws}`,
  );
});
